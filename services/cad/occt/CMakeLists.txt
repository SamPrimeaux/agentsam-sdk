cmake_minimum_required(VERSION 3.20)
project(agentsam_occt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(AGENTSAM_OCCT_PREFIX "" CACHE PATH "Installed Open CASCADE prefix (include/opencascade and lib)")
set(AGENTSAM_OCCT_TBB_PREFIX "" CACHE PATH "Optional TBB prefix for locally staged toolchains")
set(AGENTSAM_OCCT_FREETYPE_PREFIX "" CACHE PATH "Optional FreeType runtime prefix for relocated OCCT dependencies")
if(NOT AGENTSAM_OCCT_PREFIX)
  message(FATAL_ERROR "Pass -DAGENTSAM_OCCT_PREFIX=... (discover first; no implicit engine installation)")
endif()
if(NOT EXISTS "${AGENTSAM_OCCT_PREFIX}/include/opencascade/BRepPrimAPI_MakeBox.hxx")
  message(FATAL_ERROR "OCCT developer headers unavailable at supplied prefix")
endif()

add_executable(agentsam-occt main.cpp)
target_include_directories(agentsam-occt PRIVATE "${AGENTSAM_OCCT_PREFIX}/include/opencascade")
foreach(component TKDESTEP TKXSBase TKBO TKPrim TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
  find_library(_lib_${component} NAMES ${component} PATHS "${AGENTSAM_OCCT_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(agentsam-occt PRIVATE "${_lib_${component}}")
endforeach()
set(_rpaths "${AGENTSAM_OCCT_PREFIX}/lib")
if(AGENTSAM_OCCT_TBB_PREFIX)
  list(APPEND _rpaths "${AGENTSAM_OCCT_TBB_PREFIX}/lib")
endif()
if(AGENTSAM_OCCT_FREETYPE_PREFIX)
  if(NOT EXISTS "${AGENTSAM_OCCT_FREETYPE_PREFIX}/lib")
    message(FATAL_ERROR "FreeType runtime lib directory unavailable at supplied prefix")
  endif()
  list(APPEND _rpaths "${AGENTSAM_OCCT_FREETYPE_PREFIX}/lib")
endif()
set_target_properties(agentsam-occt PROPERTIES BUILD_RPATH "${_rpaths}")
